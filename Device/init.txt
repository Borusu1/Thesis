#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <SPI.h>
#include <SD.h>
#include <Adafruit_GFX.h>
#include <Adafruit_ILI9341.h>
#include <Adafruit_PN532.h>

// ---------- Wi-Fi ----------
const char* WIFI_SSID = "Bodnar_iot";
const char* WIFI_PASS = "0664995352";

// ======================================================
//  SPI #1 -> TFT + PN532
// ======================================================
#define TFT_CS     4
#define TFT_RST    5
#define TFT_DC     6
#define TFT_MOSI   7
#define TFT_SCK   15
#define TFT_MISO  16

#define PN532_SS  17

// ======================================================
//  SPI #2 -> SD only
// ======================================================
#define SD_CS     14
#define SD_MOSI   11
#define SD_SCK    12
#define SD_MISO   13

SPIClass sdSPI(HSPI);

// ---------- Button / LED ----------
#define BUTTON_PIN  9
#define LED_PIN    10

// ---------- Display ----------
Adafruit_ILI9341 tft = Adafruit_ILI9341(TFT_CS, TFT_DC, TFT_RST);

// ---------- NFC ----------
Adafruit_PN532 nfc(PN532_SS);

// ---------- Weather API ----------
const char* WEATHER_URL =
  "https://api.open-meteo.com/v1/forecast"
  "?latitude=49.8397&longitude=24.0297"
  "&current=temperature_2m,relative_humidity_2m,wind_speed_10m,weather_code"
  "&timezone=auto";

// ---------- RFID mode ----------
enum Mode {
  MODE_WRITE = 0,
  MODE_READ  = 1
};

Mode currentMode = MODE_WRITE;

// ---------- Button state ----------
bool lastButtonState = HIGH;
unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 180;

// ---------- NFC data pages ----------
const uint8_t START_PAGE = 4;
const uint8_t END_PAGE   = 11;   // 8 pages * 4 bytes = 32 bytes

// ---------- Tag anti-repeat ----------
bool tagPresentLatch = false;
uint8_t lastUid[7] = {0};
uint8_t lastUidLength = 0;
unsigned long lastTagProcessTime = 0;
const unsigned long sameTagCooldownMs = 1200;

// ---------- UI/weather ----------
String lastWeatherText = "";
unsigned long lastWeatherFetchMs = 0;
const unsigned long WEATHER_FETCH_INTERVAL = 60000;

bool showingTempMessage = false;
unsigned long tempMessageUntilMs = 0;

// ---------- Helpers ----------
void deselectMainSPI() {
  digitalWrite(TFT_CS, HIGH);
  digitalWrite(PN532_SS, HIGH);
}

void selectDisplay() {
  digitalWrite(PN532_SS, HIGH);
  digitalWrite(TFT_CS, LOW);
}

String modeText() {
  return currentMode == MODE_WRITE ? "WRITE" : "READ";
}

String weatherCodeToText(int code) {
  switch (code) {
    case 0: return "Clear";
    case 1:
    case 2:
    case 3: return "Cloudy";
    case 45:
    case 48: return "Fog";
    case 51:
    case 53:
    case 55: return "Drizzle";
    case 61:
    case 63:
    case 65: return "Rain";
    case 71:
    case 73:
    case 75: return "Snow";
    case 95: return "Thunderstorm";
    default: return "Unknown";
  }
}

String generateRandomData() {
  const char chars[] = "0123456789abcdef";
  char out[33];

  for (int i = 0; i < 32; i++) {
    out[i] = chars[esp_random() & 0x0F];
  }
  out[32] = '\0';

  return String(out);
}

bool sameUID(const uint8_t *a, uint8_t aLen, const uint8_t *b, uint8_t bLen) {
  if (aLen != bLen) return false;
  for (uint8_t i = 0; i < aLen; i++) {
    if (a[i] != b[i]) return false;
  }
  return true;
}

void blinkLed(uint8_t times, uint16_t onMs = 120, uint16_t offMs = 120) {
  for (uint8_t i = 0; i < times; i++) {
    digitalWrite(LED_PIN, HIGH);
    delay(onMs);
    digitalWrite(LED_PIN, LOW);
    if (i < times - 1) delay(offMs);
  }
}

// ---------- TFT ----------
void drawHeader(const char* title) {
  selectDisplay();
  tft.fillScreen(ILI9341_BLACK);
  tft.fillRect(0, 0, 320, 30, ILI9341_BLUE);
  tft.setTextColor(ILI9341_WHITE);
  tft.setTextSize(2);
  tft.setCursor(10, 8);
  tft.print(title);
}

void drawFooterMode() {
  selectDisplay();
  tft.fillRect(0, 220, 320, 20, ILI9341_DARKGREY);
  tft.setCursor(10, 223);
  tft.setTextColor(ILI9341_WHITE);
  tft.setTextSize(1);
  tft.print("RFID mode: ");
  tft.print(modeText());
}

void drawStatus(const String& msg, uint16_t color = ILI9341_WHITE) {
  selectDisplay();
  tft.fillRect(0, 40, 320, 175, ILI9341_BLACK);
  tft.setTextColor(color);
  tft.setTextSize(2);
  tft.setCursor(10, 50);
  tft.println(msg);
  drawFooterMode();
}

void showTextOnTFT(const char* title, const String& text, uint16_t color = ILI9341_WHITE) {
  selectDisplay();
  drawHeader(title);

  tft.setTextColor(color);
  tft.setTextSize(2);

  int y = 45;
  int start = 0;

  while (start < text.length()) {
    int next = text.indexOf('\n', start);
    if (next == -1) next = text.length();

    String line = text.substring(start, next);
    line.trim();

    if (line.length() > 0) {
      tft.setCursor(10, y);
      tft.println(line);
      y += 25;
    }

    start = next + 1;
    if (y > 210) break;
  }

  drawFooterMode();
}

void showTemporaryMessage(const char* title, const String& text, uint16_t color, unsigned long durationMs = 5000) {
  showTextOnTFT(title, text, color);
  showingTempMessage = true;
  tempMessageUntilMs = millis() + durationMs;
}

void showWeatherScreen() {
  if (lastWeatherText.length() == 0) {
    showTextOnTFT("Weather", "No weather data", ILI9341_RED);
    return;
  }
  showTextOnTFT("Weather", lastWeatherText, ILI9341_WHITE);
}

// ---------- Wi-Fi ----------
void connectWiFi() {
  drawHeader("WiFi connect");
  drawStatus("Connecting...");

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);

  int dots = 0;
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");

    selectDisplay();
    tft.fillRect(10, 90, 300, 30, ILI9341_BLACK);
    tft.setCursor(10, 90);
    tft.setTextColor(ILI9341_YELLOW);
    tft.setTextSize(2);
    tft.print("Please wait");
    for (int i = 0; i < dots; i++) tft.print(".");
    dots = (dots + 1) % 4;
  }

  Serial.println();
  Serial.println("WiFi connected");
  Serial.println(WiFi.localIP());

  selectDisplay();
  tft.fillRect(0, 40, 320, 200, ILI9341_BLACK);
  tft.setCursor(10, 50);
  tft.setTextColor(ILI9341_GREEN);
  tft.setTextSize(2);
  tft.println("WiFi connected!");
  tft.setTextColor(ILI9341_WHITE);
  tft.setCursor(10, 90);
  tft.print("IP:");
  tft.setCursor(10, 115);
  tft.println(WiFi.localIP());
  drawFooterMode();

  delay(1500);
}

// ---------- SD on separate SPI ----------
bool initSD() {
  pinMode(SD_CS, OUTPUT);
  digitalWrite(SD_CS, HIGH);

  sdSPI.begin(SD_SCK, SD_MISO, SD_MOSI, SD_CS);
  delay(10);

  if (!SD.begin(SD_CS, sdSPI, 4000000)) {
    Serial.println("SD init failed!");
    return false;
  }

  uint64_t cardSize = SD.cardSize() / (1024 * 1024);
  Serial.printf("SD OK, size: %llu MB\n", cardSize);
  return true;
}

bool saveToSD(const String& text) {
  SD.remove("/weather.txt");

  File file = SD.open("/weather.txt", FILE_WRITE);
  if (!file) {
    Serial.println("Cannot open /weather.txt for writing");
    return false;
  }

  file.println(text);
  file.close();

  Serial.println("Saved to SD: /weather.txt");
  return true;
}

bool readFromSD(String &text) {
  File file = SD.open("/weather.txt", FILE_READ);
  if (!file) {
    Serial.println("Cannot open /weather.txt for reading");
    return false;
  }

  text = "";
  while (file.available()) {
    text += (char)file.read();
  }
  file.close();

  Serial.println("Read from SD:");
  Serial.println(text);

  return true;
}

// ---------- Weather ----------
bool fetchWeather(String &resultText) {
  if (WiFi.status() != WL_CONNECTED) {
    resultText = "WiFi not connected";
    return false;
  }

  HTTPClient http;
  http.begin(WEATHER_URL);
  int httpCode = http.GET();

  if (httpCode <= 0) {
    resultText = "HTTP error";
    http.end();
    return false;
  }

  if (httpCode != 200) {
    resultText = "HTTP code: " + String(httpCode);
    http.end();
    return false;
  }

  String payload = http.getString();
  http.end();

  Serial.println("API response:");
  Serial.println(payload);

  DynamicJsonDocument doc(4096);
  DeserializationError err = deserializeJson(doc, payload);
  if (err) {
    resultText = "JSON parse failed";
    return false;
  }

  float temp = doc["current"]["temperature_2m"] | 0.0;
  int humidity = doc["current"]["relative_humidity_2m"] | 0;
  float wind = doc["current"]["wind_speed_10m"] | 0.0;
  int code = doc["current"]["weather_code"] | -1;
  const char* timeStr = doc["current"]["time"] | "unknown";

  String condition = weatherCodeToText(code);

  resultText =
    "Lviv weather\n"
    "Time: " + String(timeStr) + "\n" +
    "Temp: " + String(temp, 1) + " C\n" +
    "Humidity: " + String(humidity) + " %\n" +
    "Wind: " + String(wind, 1) + " km/h\n" +
    "Cond: " + condition;

  return true;
}

void refreshWeatherIfNeeded(bool forceNow = false) {
  if (!forceNow && millis() - lastWeatherFetchMs < WEATHER_FETCH_INTERVAL) return;

  String weatherText;
  bool apiOk = fetchWeather(weatherText);

  if (apiOk) {
    lastWeatherText = weatherText;
    saveToSD(weatherText);

    String fileText;
    if (readFromSD(fileText)) {
      lastWeatherText = fileText;
    }
  } else {
    lastWeatherText = weatherText;
  }

  lastWeatherFetchMs = millis();

  if (!showingTempMessage) {
    showWeatherScreen();
  }
}

// ---------- NFC ----------
bool initPN532() {
  deselectMainSPI();
  delay(20);

  nfc.begin();
  delay(50);

  uint32_t versiondata = nfc.getFirmwareVersion();
  if (!versiondata) {
    Serial.println("PN532 not found");
    return false;
  }

  Serial.print("Found PN532, firmware: ");
  Serial.print((versiondata >> 16) & 0xFF, DEC);
  Serial.print(".");
  Serial.println((versiondata >> 8) & 0xFF, DEC);

  nfc.SAMConfig();
  nfc.setPassiveActivationRetries(0x01);
  return true;
}

bool readPageWithRetry(uint8_t page, uint8_t *buffer, uint8_t retries = 3) {
  for (uint8_t attempt = 1; attempt <= retries; attempt++) {
    if (nfc.mifareultralight_ReadPage(page, buffer)) {
      return true;
    }
    Serial.print("Read failed on page ");
    Serial.print(page);
    Serial.print(", retry ");
    Serial.println(attempt);
    delay(15);
  }
  return false;
}

bool writePageWithRetry(uint8_t page, uint8_t *buffer, uint8_t retries = 3) {
  for (uint8_t attempt = 1; attempt <= retries; attempt++) {
    if (nfc.ntag2xx_WritePage(page, buffer)) {
      return true;
    }
    Serial.print("Write failed on page ");
    Serial.print(page);
    Serial.print(", retry ");
    Serial.println(attempt);
    delay(20);
  }
  return false;
}

bool writeDataToNTAG(const String& data) {
  if (data.length() != 32) return false;

  for (uint8_t page = START_PAGE; page <= END_PAGE; page++) {
    uint8_t buffer[4];
    int base = (page - START_PAGE) * 4;

    for (int i = 0; i < 4; i++) {
      buffer[i] = (uint8_t)data[base + i];
    }

    if (!writePageWithRetry(page, buffer, 3)) {
      return false;
    }

    delay(10);
  }

  return true;
}

bool readDataFromNTAG(String& data) {
  char result[33];
  int pos = 0;

  for (uint8_t page = START_PAGE; page <= END_PAGE; page++) {
    uint8_t buffer[4];

    if (!readPageWithRetry(page, buffer, 3)) {
      return false;
    }

    for (int i = 0; i < 4; i++) {
      result[pos++] = (char)buffer[i];
    }

    delay(5);
  }

  result[32] = '\0';
  data = String(result);
  return true;
}

bool writeAndVerify(const String& data) {
  if (!writeDataToNTAG(data)) return false;

  delay(30);

  String verifyData;
  if (!readDataFromNTAG(verifyData)) {
    Serial.println("Verify read failed");
    return false;
  }

  Serial.print("Verify read: ");
  Serial.println(verifyData);

  return verifyData == data;
}

void switchMode() {
  currentMode = (currentMode == MODE_WRITE) ? MODE_READ : MODE_WRITE;

  Serial.println();
  Serial.print("Mode switched to: ");
  Serial.println(modeText());

  if (!showingTempMessage) {
    showWeatherScreen();
  } else {
    drawFooterMode();
  }
}

void handleButton() {
  bool currentButtonState = digitalRead(BUTTON_PIN);

  if (lastButtonState == HIGH && currentButtonState == LOW) {
    if (millis() - lastDebounceTime > debounceDelay) {
      lastDebounceTime = millis();
      switchMode();
    }
  }

  lastButtonState = currentButtonState;
}

void printUID(const uint8_t *uid, uint8_t uidLength) {
  Serial.print("Tag UID: ");
  for (uint8_t i = 0; i < uidLength; i++) {
    if (uid[i] < 0x10) Serial.print("0");
    Serial.print(uid[i], HEX);
    Serial.print(" ");
  }
  Serial.println();
}

void processTag(const uint8_t *uid, uint8_t uidLength) {
  printUID(uid, uidLength);

  if (uidLength != 7) {
    Serial.println("This sketch expects NTAG / Ultralight-like tags.");
    showTemporaryMessage("RFID error", "Unsupported tag type", ILI9341_RED, 3000);
    return;
  }

  if (currentMode == MODE_WRITE) {
    String randomData = generateRandomData();

    Serial.print("Generated data: ");
    Serial.println(randomData);

    if (writeAndVerify(randomData)) {
      Serial.println("Write OK + verify OK");
      blinkLed(2);
      showTemporaryMessage("RFID written", randomData, ILI9341_GREEN, 5000);
    } else {
      Serial.println("Write FAILED or verify FAILED");
      showTemporaryMessage("Write failed", "Cannot write tag", ILI9341_RED, 3000);
    }
  } else {
    String readData;
    if (readDataFromNTAG(readData)) {
      Serial.print("Read data: ");
      Serial.println(readData);
      Serial.println("Read OK");
      blinkLed(1);
      showTemporaryMessage("RFID read", readData, ILI9341_CYAN, 5000);
    } else {
      Serial.println("Read FAILED");
      showTemporaryMessage("Read failed", "Cannot read tag", ILI9341_RED, 3000);
    }
  }
}

void handleTag() {
  uint8_t uid[7];
  uint8_t uidLength = 0;

  deselectMainSPI();
  bool found = nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, uid, &uidLength, 30);

  if (!found) {
    tagPresentLatch = false;
    lastUidLength = 0;
    return;
  }

  if (tagPresentLatch && sameUID(uid, uidLength, lastUid, lastUidLength)) {
    return;
  }

  if (millis() - lastTagProcessTime < sameTagCooldownMs &&
      sameUID(uid, uidLength, lastUid, lastUidLength)) {
    return;
  }

  memcpy(lastUid, uid, uidLength);
  lastUidLength = uidLength;
  tagPresentLatch = true;
  lastTagProcessTime = millis();

  processTag(uid, uidLength);
}

// ---------- Setup / Loop ----------
void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(TFT_CS, OUTPUT);
  pinMode(PN532_SS, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(LED_PIN, OUTPUT);

  digitalWrite(LED_PIN, LOW);
  deselectMainSPI();

  // Основна SPI для TFT + PN532
  SPI.begin(TFT_SCK, TFT_MISO, TFT_MOSI, PN532_SS);

  tft.begin();
  tft.setRotation(1);

  drawHeader("System start");
  drawStatus("Init display...", ILI9341_GREEN);
  delay(800);

  drawHeader("PN532 init");
  if (initPN532()) {
    drawStatus("PN532 OK", ILI9341_GREEN);
  } else {
    drawStatus("PN532 FAILED", ILI9341_RED);
    return;
  }
  delay(1000);

  drawHeader("SD init");
  if (initSD()) {
    drawStatus("SD card OK", ILI9341_GREEN);
  } else {
    drawStatus("SD init FAILED", ILI9341_RED);
    return;
  }
  delay(1000);

  connectWiFi();

  refreshWeatherIfNeeded(true);
  showWeatherScreen();
}

void loop() {
  handleButton();
  handleTag();
  refreshWeatherIfNeeded(false);

  if (showingTempMessage && millis() > tempMessageUntilMs) {
    showingTempMessage = false;
    showWeatherScreen();
  }

  delay(5);
}
